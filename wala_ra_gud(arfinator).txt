# 1. Inheritance (Is-a)
class Animal:
    def __init__(self, name):
        self.name = name

class Dog(Animal):
    def bark(self):
        return f"{self.name} says Woof!"

my_dog = Dog("Buddy")
print(my_dog.bark())


# 2. Composition (Strong Has-a)
class Room:
    def __init__(self, name):
        self.name = name

class House:
    def __init__(self):
        self.rooms = [Room("Kitchen"), Room("Bedroom")]

my_house = House()
for room in my_house.rooms:
    print(room.name)


# 3. Aggregation (Weak Has-a)
class Book:
    def __init__(self, title):
        self.title = title

class Library:
    def __init__(self, books=None):
        self.books = books if books else []

b1 = Book("The Hobbit")
b2 = Book("1984")
lib = Library([b1, b2])

for b in lib.books:
    print(b.title)


# 4. Association (Knows-a)
class Customer:
    def __init__(self, name):
        self.name = name
        self.orders = []

class Order:
    def __init__(self, order_id, customer):
        self.order_id = order_id
        self.customer = customer
        customer.orders.append(self)

c = Customer("Alice")
o1 = Order(101, c)
o2 = Order(102, c)

print(c.name, "orders:")
for order in c.orders:
    print(order.order_id)


# 5. Dependency (Uses-a)
class Printer:
    def print_page(self, text):
        print(text)

class Report:
    def __init__(self, content):
        self.content = content

    def generate(self, printer):
        printer.print_page(self.content)

p = Printer()
r = Report("Hello World")
r.generate(p)


# 6. Realization / Implementation (Implements)
from abc import ABC, abstractmethod

class PaymentProcessor(ABC):
    @abstractmethod
    def process_payment(self, amount):
        pass

class GCashPayment(PaymentProcessor):
    def process_payment(self, amount):
        print(f"Paid {amount} using GCash")
        return True

pay = GCashPayment()
pay.process_payment(250.75)
